/*
* Appelé par le game engine quand le joueur doit jouer 
*/
void Player::play()
{

	std::array<int,2> ar = choose_cardNumber_pos();

	owningGame->play_card(ar[0],ar[1], playerID);

	int card_drawn = owningGame->draw_card(playerID);

	// Enlève de la main du plateau pour ML
	extended_board_ml(ar[0], 9) = 0;

	if (card_drawn == -1) cards.del(ar[0]);
	else {
		remaining_cards.erase(card_drawn);
		extended_board_ml(card_drawn + 55, 9) = 0;

		cards.replace(ar[0], card_drawn);
		// Enlève du deck
		extended_board_ml(card_drawn, 9) = 1;
	};

	// Ajoute la carte sur le terrain
	extended_board_ml(ar[0], ar[1]) = 1;

	//owningGame->end_turn(playerID);

	return;

}

std::array<int, 2> Player::choose_cardNumber_pos() {
	std::uniform_int_distribution<> distrib(0, cards.size - 1);
	return std::array<int, 2>(cards.data[distrib(gen)],0);
};

/*
* Met à jour les infos du terrain quand l'adversaire a joué une carte
* Carte restante dans un 
*/
void Player::update_adversary_card_played(int card, int n_frontier, PlayerType player)
{

	extended_board_ml(card+55, n_frontier) = 1;
	remaining_cards.erase(card);
	extended_board_ml(card + 55, 9) = 0;

	int unknown_cards_size = remaining_cards.size();
	// p / k
	int taille_cartes_adversaire = 6;
	if (unknown_cards_size < 6) {
		taille_cartes_adversaire = unknown_cards_size;
	}

	for (auto it = remaining_cards.begin(); it != remaining_cards.end(); it++) {
		extended_board_ml(*it + 55, 9) = taille_cartes_adversaire / unknown_cards_size;
	}
	


}

{

}
